   309	function getDealDecision({
   310	  score,
   311	  roi,
   312	  profit,
   313	  executiveScore,
   314	  successProbability,
   315	  buySignal,
   316	  allocationScore,
   317	  allocationTier,
   318	}) {
   319	  if (
   320	    allocationTier === "TIER_1" ||
   321	    allocationScore >= 85 ||
   322	    buySignal === "FUERTE" ||
   323	    (successProbability >= 85 && executiveScore >= 80 && profit >= 2500)
   324	  ) {
   325	    return "Priorizar capital";
   326	  }
   327	
   328	  if (
   329	    allocationTier === "TIER_2" ||
   330	    allocationScore >= 70 ||
   331	    buySignal === "INTERESANTE" ||
   332	    (successProbability >= 72 && executiveScore >= 70)
   333	  ) {
   334	    return "Comprar selectivo";
   335	  }
   336	
   337	  if (
   338	    allocationTier === "TIER_3" ||
   339	    allocationScore >= 55 ||
   340	    score >= 65 ||
   341	    executiveScore >= 65 ||
   342	    successProbability >= 58
   343	  ) {
   344	    return "Vigilar";
   345	  }
   346	
   347	  return "Descartar";
   348	}

   350	function buildDecisionText({
   351	  score,
   352	  roi,
   353	  profit,
   354	  priorityScore,
   355	  executiveScore,
   356	  learningBonus,
   357	  historicalConfidence,
   358	  successProbability,
   359	  buySignal,
   360	  estimatedSellDays,
   361	  expectedROI,
   362	  expectedProfit,
   363	  allocationScore,
   364	  allocationTier,
   365	  marketTimingLabel,
   366	}) {
   367	  const learningText =
   368	    learningBonus !== 0
   369	      ? ` Aprendizaje histórico aplicado: ${formatBonus(learningBonus)}.`
   370	      : "";
   371	
   372	  if (allocationTier === "TIER_1" || allocationScore >= 85) {
   373	    return `Prioridad máxima de capital: allocation score ${allocationScore}/100, tier ${allocationTier}, timing ${marketTimingLabel}, probabilidad de éxito ${successProbability}/100, venta estimada en ${estimatedSellDays} días, ROI esperado ${expectedROI}% y beneficio esperado de ${expectedProfit} €.${learningText}`;
   374	  }
   375	
   376	  if (allocationTier === "TIER_2" || allocationScore >= 70) {
   377	    return `Buena asignación potencial: allocation score ${allocationScore}/100, tier ${allocationTier}, probabilidad de éxito ${successProbability}/100, ROI ${roi}% y beneficio estimado de ${profit} €.${learningText}`;
   378	  }
   379	
   380	  if (buySignal === "FUERTE" || successProbability >= 85) {
   381	    return `Señal fuerte: probabilidad de éxito ${successProbability}/100, venta estimada en ${estimatedSellDays} días, ROI esperado ${expectedROI}% y beneficio esperado de ${expectedProfit} €. Executive score ${executiveScore}, score IA ${score}, confianza histórica ${historicalConfidence}.${learningText}`;
   382	  }
   383	
   384	  if (buySignal === "INTERESANTE" || successProbability >= 72) {
   385	    return `Oportunidad interesante: probabilidad de éxito ${successProbability}/100, prioridad ${priorityScore}, ROI ${roi}% y beneficio estimado de ${profit} €.${learningText}`;
   386	  }
   387	
   388	  if (score >= 65 || executiveScore >= 65 || successProbability >= 58) {
   389	    return `Oportunidad intermedia: conviene vigilar precio, demanda, margen, timing y señales históricas.${learningText}`;
   390	  }
   391	
   392	  return `Oportunidad débil dentro del ranking. No priorizar capital salvo contexto especial.${learningText}`;
   393	}

   395	function getDecisionMode(items = []) {
   396	  const tierOneCount = items.filter((item) => {
   397	    return item.allocationTier === "TIER_1" || Number(item.allocationScore || 0) >= 85;
   398	  }).length;
   399	
   400	  const tierTwoCount = items.filter((item) => {
   401	    return item.allocationTier === "TIER_2" || Number(item.allocationScore || 0) >= 70;
   402	  }).length;
   403	
   404	  if (tierOneCount >= 2) return "Asignar capital";
   405	  if (tierOneCount >= 1) return "Priorizar una";
   406	  if (tierTwoCount >= 2) return "Comprar selectivo";
   407	  if (items.length >= 5) return "Analizar";
   408	  return "Esperar datos";
   409	}

   411	function getRankingQuality(score = 0) {

   413

   415	  if (value >= 65) return "Media";
   416	  if (value > 0) return "Inicial";
   417	  return "Sin datos";
   418	}
