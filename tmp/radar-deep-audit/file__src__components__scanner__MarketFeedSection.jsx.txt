   401	function buildRadarHealthRecommendation(rejectionSummary) {
   402	  if (!rejectionSummary || typeof rejectionSummary !== "object") {
   403	    return "";
   404	  }
   405	
   406	  const [mainReason, count] =
   407	    Object.entries(rejectionSummary).sort((a, b) => b[1] - a[1])[0] || [];
   408	
   409	  if (!mainReason || !count) return "";
   410	
   411	  if (mainReason === "modelo_incompatible") {
   412	    return `El filtro de modelo está bloqueando ${count} anuncios. Revisa compatibilidad y catálogo antes de tocar ranking.`;
   413	  }
   414	
   415	  if (mainReason === "modelo_no_confirmado") {
   416	    return `Hay ${count} anuncios donde el modelo no queda confirmado. El feed puede estar trayendo resultados útiles con títulos incompletos.`;
   417	  }
   418	
   419	  if (mainReason === "precio_sobre_presupuesto") {
   420	    return `El presupuesto está bloqueando ${count} anuncios. Puede convenir mostrar descartados caros o relajar el rango.`;
   421	  }
   422	
   423	  if (mainReason === "km_alto") {
   424	    return `El kilometraje está bloqueando ${count} anuncios. Revisa si el límite de km es demasiado estricto para este mercado.`;
   425	  }
   426	
   427	  if (mainReason === "fuel_no_phev") {
   428	    return `El filtro PHEV está bloqueando ${count} anuncios. Revisa detección de combustible antes de descartar oportunidades.`;
   429	  }
   430	
   431	  if (mainReason === "score_bajo") {
   432	    return `El Radar encuentra mercado, pero ${count} anuncios no alcanzan score suficiente. Revisa scoring y filtros de compatibilidad.`;
   433	  }
   434	
   435	  return `Motivo dominante: ${mainReason} (${count}). Prioriza revisar ese filtro antes de añadir más motores.`;
   436	}

   438	function getDiagnosticsList(realFeedDiagnostics) {
   439	  if (Array.isArray(realFeedDiagnostics)) {
   440	    return realFeedDiagnostics;
   441	  }
   442	
   443	  if (!realFeedDiagnostics || typeof realFeedDiagnostics !== "object") {
   444	    return [];
   445	  }
   446	
   447	  return Object.values(realFeedDiagnostics).filter(
   448	    (item) =>
   449	      item &&
   450	      typeof item === "object" &&
   451	      ("source" in item || "status" in item || "parsedCount" in item)
   452	  );
   453	}

   455	function getStatusLabel(status) {
   456	  if (status === "success") return "OK";
   457	  if (status === "error") return "🚫 Bloqueado por la fuente";
   458	  if (status === "no-results") return "Sin anuncios";
   459	  return status || "Sin dato";
