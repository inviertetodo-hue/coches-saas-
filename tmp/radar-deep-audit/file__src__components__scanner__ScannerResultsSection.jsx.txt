    41	function SaveBestOpportunityPanel({ marketFeed, scan }) {
    42	  const [saving, setSaving] = useState(false);
    43	  const [saved, setSaved] = useState(false);
    44	  const [message, setMessage] = useState("");
    45	
    46	  const candidate = useMemo(() => {
    47	    const opportunities = marketFeed?.opportunities || [];
    48	
    49	    return (
    50	      opportunities.find((item) => {
    51	        const semanticScore = Number(item.semanticScore || 0);
    52	        const finalScore = Number(item.finalDecision?.finalScore || 0);
    53	        const profit = Number(item.netProfit || 0);
    54	        const roi = Number(item.netRoi || 0);
    55	
    56	        return (
    57	          semanticScore >= 80 &&
    58	          finalScore >= 70 &&
    59	          Number.isFinite(profit) &&
    60	          Number.isFinite(roi) &&
    61	          item.brand &&
    62	          item.model
    63	        );
    64	      }) || null
    65	    );
    66	  }, [marketFeed]);
    67	
    68	  if (marketFeed?.sourceMode !== "real-feed") {
    69	    return null;
    70	  }
    71	
    72	  if (!candidate) {
    73	    return (
    74	      <section style={savePanelStyle}>
    75	        <p style={saveEyebrowStyle}>Inteligencia Mercado</p>
    76	        <h3 style={saveTitleStyle}>Aún no hay oportunidad suficientemente limpia para guardar</h3>
    77	        <p style={saveTextStyle}>
    78	          El Scanner ha encontrado datos reales, pero todavía no hay una oportunidad con match semántico alto, marca, modelo y score suficiente.
    79	        </p>
    80	      </section>
    81	    );
    82	  }

    84	  async function saveCandidate() {
    85	    if (saving || saved) return;
    86	
    87	    setSaving(true);
    88	    setMessage("");
    89	
    90	    const payload = buildScannerOpportunityPayload({
    91	      candidate,
    92	      scan,
    93	    });
    94	
    95	    const { error } = await supabase.from("import_analyses").insert(payload);
    96	
    97	    if (error) {
    98	      console.error("Error saving scanner opportunity:", error);
    99	      setMessage("No se ha podido guardar en Inteligencia Mercado.");
   100	      setSaving(false);
   101	      return;
   102	    }
   103	
   104	    setSaved(true);
   105	    setSaving(false);
   106	    setMessage("Oportunidad guardada. Ya alimenta Inteligencia Mercado.");
   107	  }
   108	
   109	  return (
   110	    <section style={savePanelStyle}>
   111	      <div>
   112	        <p style={saveEyebrowStyle}>Inteligencia Mercado</p>
   113	
   114	        <h3 style={saveTitleStyle}>Guardar oportunidad seleccionada</h3>
   115	
   116	        <p style={saveTextStyle}>
   117	          {candidate.title} · Score {candidate.finalDecision?.finalScore || 0}/100 · ROI {candidate.netRoi}% · Match {candidate.semanticScore || 0}/100
   118	        </p>
   119	      </div>
   120	
   121	      <button
   122	        onClick={saveCandidate}
   123	        disabled={saving || saved}
   124	        style={{
   125	          ...saveButtonStyle,
   126	          opacity: saving || saved ? 0.7 : 1,
   127	          cursor: saving || saved ? "not-allowed" : "pointer",
   128	        }}
   129	      >
   130	        {saving ? "Guardando..." : saved ? "✅ Guardado" : "💾 Guardar en Inteligencia Mercado"}
   131	      </button>
   132	
   133	      {message && <p style={saveMessageStyle}>{message}</p>}
   134	    </section>
   135	  );
   136	}

   138	function ScannerLoadingSection({ scan }) {
   139	  const [seconds, setSeconds] = useState(8);
   140	
   141	  const searchLinks = scan?.searchLinks || [];
   142	  const countries = [...new Set(searchLinks.map((item) => item.country))];
   143	
   144	  const loadingSteps = useMemo(
   145	    () => [
   146	      "Consultando mobile.de",
   147	      "Consultando AutoScout24",
   148	      "Analizando precios europeos",
   149	      "Calculando liquidez",
   150	      "Evaluando riesgo",
   151	      "Generando ranking IA",
   152	    ],
   153	    []
   154	  );
   155	
   156	  useEffect(() => {
   157	    setSeconds(8);
   158	
   159	    const interval = window.setInterval(() => {
   160	      setSeconds((current) => {
   161	        if (current <= 1) return 1;
   162	
   163	        return current - 1;
   164	      });
   165	    }, 1000);
   166	
   167	    return () => window.clearInterval(interval);
   168	  }, [scan?.query, scan?.country, scan?.maxBudget]);
   169	
   170	  const progress = Math.min(
   171	    100,
   172	    Math.max(12, Math.round(((9 - seconds) / 8) * 100))
   173	  );
   174	
   175	  const activeStepIndex = Math.min(
   176	    loadingSteps.length - 1,
   177	    Math.floor((progress / 100) * loadingSteps.length)
   178	  );
   179	
   180	  return (
   181	    <section style={loadingStyle}>
   182	      <div style={loadingIconStyle}>🌍</div>
   183	
   184	      <div>
   185	        <p style={eyebrowStyle}>Escaneo en curso</p>
   186	
   187	        <h2 style={loadingTitleStyle}>
   188	          Escaneando mercado europeo del automóvil
   189	        </h2>
   190	
   191	        <p style={loadingTextStyle}>
   192	          Analizando oportunidades, precios, liquidez, riesgo y margen potencial
   193	          en los principales mercados europeos.
   194	        </p>
   195	
   196	        <div style={progressTrackStyle}>
   197	          <div
   198	            style={{
   199	              ...progressBarStyle,
   200	              width: `${progress}%`,
   201	            }}
   202	          />
   203	        </div>
   204	
   205	        <div style={progressMetaStyle}>
   206	          <span>{progress}% completado</span>
   207	          <span>{seconds} segundos</span>
   208	        </div>
   209	
   210	        <div style={loadingStepsStyle}>
   211	          {loadingSteps.map((step, index) => (
   212	            <span
   213	              key={step}
   214	              style={{
   215	                ...stepPillStyle,
   216	                ...(index <= activeStepIndex ? stepPillActiveStyle : {}),
   217	              }}
   218	            >
   219	              {index < activeStepIndex
   220	                ? "✓"
   221	                : index === activeStepIndex
   222	                  ? "⏳"
   223	                  : "○"}{" "}
   224	              {step}
   225	            </span>
   226	          ))}
   227	        </div>
   228	      </div>
   229	
   230	      <div style={loadingStatsStyle}>
   231	        <strong style={loadingNumberStyle}>{countries.length || 1}</strong>
   232	        <span style={loadingLabelStyle}>mercados</span>
   233	
   234	        <strong style={loadingNumberStyle}>{searchLinks.length || 1}</strong>
   235	        <span style={loadingLabelStyle}>búsquedas</span>
   236	      </div>
   237	    </section>
   238	  );
