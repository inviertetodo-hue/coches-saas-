   459	function inferKw(item) {
   460	  const directKw = Number(
   461	    item.kw ||
   462	      item.powerKw ||
   463	      item.power_kw ||
   464	      item.engineKw ||
   465	      item.engine_kw ||
   466	      item.power
   467	  );
   468	
   469	  if (Number.isFinite(directKw) && directKw > 40 && directKw < 600) {
   470	    return directKw;
   471	  }
   472	
   473	  const text = normalizeForModelRule(
   474	    [item.title, item.engine, item.description, item.power].join(" ")
   475	  );
   476	
   477	  const kwMatch = text.match(/\b(\d{2,3})\s?kw\b/);
   478	  if (kwMatch) return Number(kwMatch[1]);
   479	
   480	  const cvMatch = text.match(/\b(\d{2,3})\s?(cv|ps|hp)\b/);
   481	  if (cvMatch) return Math.round(Number(cvMatch[1]) * 0.7355);
   482	
   483	  return NaN;
   484	}

   539	async function saveBestRealOpportunityToMarketMemory({
   540	  sourceMode,
   541	  opportunities,
   542	  scan,
   543	  savedScanRef,
   544	}) {
   545	  if (sourceMode !== "real-feed") return;
   546	  if (!Array.isArray(opportunities) || opportunities.length === 0) return;
   547	
   548	  const bestCandidate = opportunities.find((item) => {
   549	    const semanticScore = Number(item.semanticScore || 0);
   550	    const finalScore = Number(item.finalDecision?.finalScore || 0);
   551	    const profit = Number(item.netProfit || 0);
   552	    const roi = Number(item.netRoi || 0);
   553	
   554	    return (
   555	      semanticScore >= 80 &&
   556	      finalScore >= 70 &&
   557	      Number.isFinite(profit) &&
   558	      Number.isFinite(roi)
   559	    );
   560	  });
   561	
   562	  if (!bestCandidate) return;
   563	
   564	  const saveKey = [
   565	    normalizeForKey(scan?.query),
   566	    normalizeForKey(bestCandidate.title),
   567	    bestCandidate.price,
   568	    bestCandidate.km || bestCandidate.mileage,
   569	    bestCandidate.year,
   570	    bestCandidate.country,
   571	  ].join("|");
   572	
   573	  if (!saveKey || savedScanRef.current === saveKey) return;
   574	
   575	  const sessionKey = `scanner-memory:${saveKey}`;
   576	
   577	  if (window.sessionStorage.getItem(sessionKey)) return;
   578	
   579	  savedScanRef.current = saveKey;
   580	  window.sessionStorage.setItem(sessionKey, "saved");
   581	
   582	  const {
   583	    data: { user },
   584	  } = await supabase.auth.getUser();
   585	
   586	  if (!user) {
   587	    return;
   588	  }
   589	
   590	  const payload = buildScannerOpportunityPayload({
   591	    candidate: bestCandidate,
   592	    scan,
   593	  });
   594	
   595	  const { error } = await supabase.from("import_analyses").insert(payload);
   596	
   597	  if (error) {
   598	    console.error("Error saving scanner opportunity:", error);
   599	    window.sessionStorage.removeItem(sessionKey);
   600	    savedScanRef.current = "";
   601	  }
   602	}
