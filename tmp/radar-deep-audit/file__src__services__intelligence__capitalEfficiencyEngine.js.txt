     1	export function buildCapitalEfficiency({
     2	  price = 0,
     3	  profit = 0,
     4	  roi = 0,
     5	  estimatedSellDays = 60,
     6	  successProbability = 0,
     7	  executiveBuySignalScore = 0,
     8	} = {}) {
     9	  const capitalRequired = safeNumber(price);
    10	  const expectedProfit = safeNumber(profit);
    11	  const expectedROI = safeNumber(roi);
    12	  const sellDays = Math.max(safeNumber(estimatedSellDays), 1);
    13	  const successScore = safeNumber(successProbability);
    14	  const executiveScore = safeNumber(executiveBuySignalScore);
    15	
    16	  const capitalVelocity = calculateCapitalVelocity(sellDays);
    17	  const profitPerDay = expectedProfit / sellDays;
    18	  const profitPerThousand = capitalRequired > 0
    19	    ? (expectedProfit / capitalRequired) * 1000
    20	    : 0;
    21	
    22	  const annualizedProfitPotential = Math.round(profitPerDay * 365);
    23	
    24	  let capitalEfficiencyScore = 0;
    25	
    26	  capitalEfficiencyScore += Math.min(expectedROI * 2.2, 28);
    27	  capitalEfficiencyScore += Math.min(profitPerThousand * 2.6, 24);
    28	  capitalEfficiencyScore += capitalVelocity * 0.22;
    29	  capitalEfficiencyScore += successScore * 0.18;
    30	  capitalEfficiencyScore += executiveScore * 0.12;
    31	
    32	  capitalEfficiencyScore = clamp(
    33	    Math.round(capitalEfficiencyScore),
    34	    0,
    35	    100
    36	  );
    37	
    38	  return {
    39	    capitalRequired,
    40	    expectedProfit,
    41	    expectedROI,
    42	    estimatedSellDays: sellDays,
    43	    profitPerDay: round(profitPerDay),
    44	    profitPerThousand: round(profitPerThousand),
    45	    capitalVelocity,
    46	    annualizedProfitPotential,
    47	    capitalEfficiencyScore,
    48	    capitalEfficiencyLabel: getCapitalEfficiencyLabel(capitalEfficiencyScore),
    49	    capitalRisk: getCapitalRisk({
    50	      capitalRequired,
    51	      sellDays,
    52	      successScore,
    53	    }),
    54	    summary: buildSummary({
    55	      capitalEfficiencyScore,
    56	      capitalRequired,
    57	      expectedProfit,
    58	      sellDays,
    59	      annualizedProfitPotential,
    60	    }),
    61	  };
    62	}
    63	
    64	function calculateCapitalVelocity(days) {
    65	  if (days <= 14) return 100;
    66	  if (days <= 21) return 92;
    67	  if (days <= 30) return 82;
    68	  if (days <= 45) return 68;
    69	  if (days <= 60) return 55;
    70	  if (days <= 90) return 38;
    71	  return 20;
    72	}

    74	function getCapitalEfficiencyLabel(score) {
    75	  if (score >= 85) return "CAPITAL_STAR";
    76	  if (score >= 72) return "EFFICIENT";
    77	  if (score >= 58) return "ACCEPTABLE";
    78	  if (score >= 42) return "WEAK";
    79	  return "POOR";
    80	}
    81	
    82	function getCapitalRisk({
    83	  capitalRequired,
    84	  sellDays,
    85	  successScore,
    86	}) {
    87	  if (capitalRequired >= 70000 && sellDays >= 60) return "Alto";
    88	  if (successScore < 55 && sellDays >= 60) return "Alto";
    89	  if (capitalRequired >= 50000 && sellDays >= 45) return "Medio-alto";
    90	  if (sellDays <= 30 && successScore >= 75) return "Bajo";
    91	  return "Medio";
    92	}
    93	
    94	function buildSummary({
    95	  capitalEfficiencyScore,
    96	  capitalRequired,
    97	  expectedProfit,
    98	  sellDays,
    99	  annualizedProfitPotential,
   100	}) {
   101	  return `Eficiencia de capital ${capitalEfficiencyScore}/100: requiere ${formatMoney(
   102	    capitalRequired
   103	  )}, beneficio esperado ${formatMoney(
   104	    expectedProfit
   105	  )}, rotación estimada ${sellDays} días y potencial anualizado ${formatMoney(
   106	    annualizedProfitPotential
   107	  )}.`;
   108	}
   109	
   110	function formatMoney(value) {
   111	  return `${safeNumber(value).toLocaleString("es-ES")} €`;
   112	}
   113	
   114	function safeNumber(value) {
   115	  const number = Number(value);
   116	  return Number.isFinite(number) ? number : 0;
   117	}
   118	
   119	function round(value) {
   120	  return Math.round(value * 100) / 100;
   121	}
   122	
   123	function clamp(value, min, max) {
   124	  return Math.min(Math.max(value, min), max);
   125	}
