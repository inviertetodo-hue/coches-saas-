     5	export function buildComparableVehicles(vehicle = {}, memoryRecords = []) {
     6	  const comparables = memoryRecords
     7	    .filter((record) => isComparable(vehicle, record))
     8	    .map((record) => {
     9	      const comparableScore = calculateComparableScore(vehicle, record);
    10	
    11	      return {
    12	        ...record,
    13	        comparableScore,
    14	        comparableQualityScore: comparableScore,
    15	        comparableReasons: buildComparableReasons(vehicle, record),
    16	      };
    17	    })
    18	    .filter((record) => record.comparableScore >= MIN_COMPARABLE_SCORE)
    19	    .sort((a, b) => b.comparableScore - a.comparableScore);
    20	
    21	  const bestComparables = comparables.slice(0, 20);
    22	
    23	  return {
    24	    totalComparables: comparables.length,
    25	    bestComparables,
    26	    bestMatch: bestComparables[0] || null,
    27	    averageComparablePrice: calculateAveragePrice(bestComparables),
    28	    averageComparableKm: calculateAverageKm(bestComparables),
    29	    averageComparableYear: calculateAverageYear(bestComparables),
    30	    quality: buildComparableQuality(bestComparables),
    31	  };
    32	}

    34	function isComparable(vehicle = {}, record = {}) {
    35	  const status = String(record.status || "ACTIVE").toUpperCase();
    36	
    37	  if (status !== "ACTIVE") {
    38	    return false;
    39	  }
    40	
    41	  if (!isRealComparableRecord(record)) {
    42	    return false;
    43	  }
    44	
    45	  const vehicleBrand = normalize(vehicle.brand);
    46	  const vehicleModel = normalize(vehicle.model);
    47	
    48	  const recordBrand = normalize(record.brand);
    49	  const recordModel = normalize(record.model);
    50	
    51	  if (!vehicleBrand || !vehicleModel) {
    52	    return false;
    53	  }
    54	
    55	  if (vehicleBrand !== recordBrand) {
    56	    return false;
    57	  }
    58	
    59	  if (vehicleModel !== recordModel) {
    60	    return false;
    61	  }
    62	
    63	  const vehicleYear = toNumber(vehicle.year);
    64	  const recordYear = toNumber(record.year);
    65	
    66	  if (
    67	    vehicleYear > 0 &&
    68	    recordYear > 0 &&
    69	    Math.abs(vehicleYear - recordYear) > MAX_YEAR_DISTANCE
    70	  ) {
    71	    return false;
    72	  }
    73	
    74	  const vehicleKm = toNumber(vehicle.km ?? vehicle.mileage);
    75	  const recordKm = toNumber(record.km ?? record.mileage);
    76	
    77	  if (
    78	    vehicleKm > 0 &&
    79	    recordKm > 0 &&
    80	    Math.abs(vehicleKm - recordKm) > MAX_KM_DISTANCE
    81	  ) {
    82	    return false;
    83	  }
    84	
    85	  return true;
    86	}

    88	function isRealComparableRecord(record = {}) {
    89	  const sourceMode = normalize(record.sourceMode);
    90	
    91	  const blockedSourceModes = new Set([
    92	    "mock-fallback",
    93	    "mock-fallback-ranking",
    94	    "review",
    95	    "unresolved-url-query",
    96	    "brand-url-query",
    97	    "semantic-url-query",
    98	    "ready_for_live_feed",
    99	    "missing_model",
   100	    "insufficient_query",
   101	  ]);
   102	
   103	  if (blockedSourceModes.has(sourceMode)) {
   104	    return false;
   105	  }
   106	
   107	  if (record.needsLiveMarketFeed) {
   108	    return false;
   109	  }
   110	
   111	  if (record.isRealMarketData === false) {
   112	    return false;
   113	  }
   114	
   115	  const brand = normalize(record.brand);
   116	  const model = normalize(record.model);
   117	  const year = toNumber(record.year);
   118	  const price = toNumber(record.price);
   119	  const km = toNumber(record.km ?? record.mileage);
   120	
   121	  if (!brand || !model) {
   122	    return false;
   123	  }
   124	
   125	  if (model === "gama") {
   126	    return false;
   127	  }
   128	
   129	  if (year <= 0 || price <= 0 || km <= 0) {
   130	    return false;
   131	  }
   132	
   133	  return true;
   134	}

   136	function calculateComparableScore(vehicle = {}, record = {}) {
   137	  let score = 100;
   138	
   139	  const vehicleYear = toNumber(vehicle.year);
   140	  const recordYear = toNumber(record.year);
   141	
   142	  if (vehicleYear > 0 && recordYear > 0) {
   143	    const yearDistance = Math.abs(vehicleYear - recordYear);
   144	
   145	    score -= yearDistance * 10;
   146	
   147	    if (yearDistance === 0) score += 6;
   148	    if (yearDistance === 1) score += 3;
   149	  }
   150	
   151	  const vehicleKm = toNumber(vehicle.km ?? vehicle.mileage);
   152	  const recordKm = toNumber(record.km ?? record.mileage);
   153	
   154	  if (vehicleKm > 0 && recordKm > 0) {
   155	    const kmDistance = Math.abs(vehicleKm - recordKm);
   156	
   157	    score -= Math.floor(kmDistance / 5000);
   158	
   159	    if (kmDistance <= 15000) score += 8;
   160	    if (kmDistance > 50000) score -= 10;
   161	    if (kmDistance > 80000) score -= 10;
   162	  }
   163	
   164	  if (sameNormalized(vehicle.fuelType ?? vehicle.fuel_type, record.fuelType ?? record.fuel_type)) {
   165	    score += 5;
   166	  }
   167	
   168	  if (sameNormalized(vehicle.drivetrain, record.drivetrain)) {
   169	    score += 4;
   170	  }
   171	
   172	  if (
   173	    sameNormalized(
   174	      vehicle.performancePackage ?? vehicle.performance_package,
   175	      record.performancePackage ?? record.performance_package
   176	    )
   177	  ) {
   178	    score += 4;
   179	  }
   180	
   181	  const confidence = toNumber(record.comparableConfidence);
   182	
   183	  if (confidence >= 85) score += 8;
   184	  if (confidence >= 75) score += 5;
   185	  if (confidence > 0 && confidence < 60) score -= 8;
   186	
   187	  return clamp(score, 0, 100);
   188	}

   190	function buildComparableReasons(vehicle = {}, record = {}) {
   191	  const reasons = [];
   192	
   193	  if (normalize(vehicle.brand) === normalize(record.brand)) {
   194	    reasons.push("Misma marca");
   195	  }
   196	
   197	  if (normalize(vehicle.model) === normalize(record.model)) {
   198	    reasons.push("Mismo modelo");
   199	  }
   200	
   201	  const vehicleYear = toNumber(vehicle.year);
   202	  const recordYear = toNumber(record.year);
   203	
   204	  if (
   205	    vehicleYear > 0 &&
   206	    recordYear > 0 &&
   207	    Math.abs(vehicleYear - recordYear) <= 1
   208	  ) {
   209	    reasons.push("Año muy similar");
   210	  } else if (
   211	    vehicleYear > 0 &&
   212	    recordYear > 0 &&
   213	    Math.abs(vehicleYear - recordYear) <= MAX_YEAR_DISTANCE
   214	  ) {
   215	    reasons.push("Año comparable");
   216	  }
   217	
   218	  const vehicleKm = toNumber(vehicle.km ?? vehicle.mileage);
   219	  const recordKm = toNumber(record.km ?? record.mileage);
   220	
   221	  if (
   222	    vehicleKm > 0 &&
   223	    recordKm > 0 &&
   224	    Math.abs(vehicleKm - recordKm) <= 20000
   225	  ) {
   226	    reasons.push("Kilometraje muy similar");
   227	  } else if (
   228	    vehicleKm > 0 &&
   229	    recordKm > 0 &&
   230	    Math.abs(vehicleKm - recordKm) <= MAX_KM_DISTANCE
   231	  ) {
   232	    reasons.push("Kilometraje comparable");
   233	  }
   234	
   235	  if (sameNormalized(vehicle.fuelType ?? vehicle.fuel_type, record.fuelType ?? record.fuel_type)) {
   236	    reasons.push("Mismo combustible");
   237	  }
   238	
   239	  if (sameNormalized(vehicle.drivetrain, record.drivetrain)) {
   240	    reasons.push("Misma tracción");
   241	  }
   242	
   243	  if (
   244	    sameNormalized(
   245	      vehicle.performancePackage ?? vehicle.performance_package,
   246	      record.performancePackage ?? record.performance_package
   247	    )
   248	  ) {
   249	    reasons.push("Paquete similar");
   250	  }
   251	
   252	  return reasons;
   253	}

   295	  const values = records
   296	    .map((item) => toNumber(item.price))
   297	    .filter((value) => value > 0);
   298	
   299	  if (!values.length) {

   301	  }
   302	
   303	  return Math.round(
   304	    values.reduce((sum, value) => sum + value, 0) / values.length

   350	function sameNormalized(a, b) {
   351	  const left = normalize(a);
   352	  const right = normalize(b);
   353	
   354	  return Boolean(left && right && left === right);
   355	}
