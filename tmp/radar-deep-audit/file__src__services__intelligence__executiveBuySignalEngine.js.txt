     1	export function buildExecutiveBuySignal({
     2	  executiveScore = 0,
     3	  successProbability = 0,
     4	  confidenceScore = 0,
     5	  liquidityScore = 60,
     6	  riskScore = 50,
     7	  timelineMomentumScore = 50,
     8	  timelineMomentumLabel = "NO_HISTORY",
     9	} = {}) {
    10	  const momentumScore = normalizeMomentumScore(
    11	    timelineMomentumScore,
    12	    timelineMomentumLabel
    13	  );
    14	
    15	  const score =
    16	    executiveScore * 0.30 +
    17	    successProbability * 0.30 +
    18	    confidenceScore * 0.14 +
    19	    liquidityScore * 0.09 +
    20	    (100 - riskScore) * 0.05 +
    21	    momentumScore * 0.12;
    22	
    23	  const finalScore = clamp(Math.round(score), 0, 100);
    24	
    25	  return {
    26	    finalScore,
    27	    signal: getSignal(finalScore),
    28	    color: getColor(finalScore),
    29	    summary: buildSummary(finalScore, timelineMomentumLabel),
    30	    timelineMomentumScore: momentumScore,
    31	    timelineMomentumLabel,
    32	  };
    33	}

    35	function normalizeMomentumScore(score, label) {
    36	  const numericScore = Number(score);
    37	
    38	  if (Number.isFinite(numericScore) && numericScore > 0) {
    39	    return clamp(Math.round(numericScore), 0, 100);
    40	  }
    41	
    42	  if (label === "IMPROVING_FAST") return 88;
    43	  if (label === "IMPROVING") return 72;
    44	  if (label === "STABLE") return 52;
    45	  if (label === "DETERIORATING") return 34;
    46	  if (label === "AVOID_TREND") return 18;
    47	
    48	  return 50;
    49	}

    51	function getSignal(score) {
    52	  if (score >= 90) return "STRONG_BUY";
    53	  if (score >= 78) return "BUY";
    54	  if (score >= 62) return "WATCHLIST";
    55	  if (score >= 45) return "AVOID";
    56	  return "REJECT";
    57	}
    58	
    59	function getColor(score) {
    60	  if (score >= 90) return "green";
    61	  if (score >= 78) return "lime";
    62	  if (score >= 62) return "yellow";
    63	  if (score >= 45) return "orange";
    64	  return "red";
    65	}

    73	
    74	  if (score >= 78) {

    76	  }

    79	    return `Mantener en seguimiento. ${momentumText}`;
    80	  }

    82	  if (score >= 45) {
    83	    return `No priorizar actualmente. ${momentumText}`;
    84	  }

    89	function buildMomentumSummary(label) {

    91	    return "La evolución temporal mejora de forma clara.";
    92	  }
    93	
    94	  if (label === "IMPROVING") {
    95	    return "La evolución temporal es positiva.";
    96	  }
    97	
    98	  if (label === "STABLE") {
    99	    return "La evolución temporal es estable.";
   100	  }
   101	
   102	  if (label === "DETERIORATING") {
   103	    return "La evolución temporal se está deteriorando.";
   104	  }
   105	
   106	  if (label === "AVOID_TREND") {
   107	    return "La evolución temporal aconseja evitar la operación.";
