     1	export function buildInventoryRisk({
     2	  estimatedSellDays = 60,
     3	  successProbability = 0,
     4	  capitalEfficiencyScore = 50,
     5	  historicalConfidenceScore = 50,
     6	  price = 0,
     7	} = {}) {
     8	  const sellDays = safeNumber(estimatedSellDays);
     9	  const success = safeNumber(successProbability);
    10	  const capitalEfficiency = safeNumber(capitalEfficiencyScore);
    11	  const confidence = safeNumber(historicalConfidenceScore);
    12	  const capital = safeNumber(price);
    13	
    14	  let riskScore = 0;
    15	
    16	  riskScore += sellDays * 0.55;
    17	  riskScore += Math.max(0, 100 - success) * 0.25;
    18	  riskScore += Math.max(0, 100 - capitalEfficiency) * 0.12;
    19	  riskScore += Math.max(0, 100 - confidence) * 0.08;
    20	
    21	  if (capital >= 60000) {
    22	    riskScore += 8;
    23	  }
    24	
    25	  if (capital >= 90000) {
    26	    riskScore += 10;
    27	  }
    28	
    29	  riskScore = clamp(Math.round(riskScore), 0, 100);
    30	
    31	  return {
    32	    inventoryRiskScore: riskScore,
    33	    inventoryRiskLabel: getInventoryRiskLabel(riskScore),
    34	    immobilizedCapitalRisk: getImmobilizedCapitalRisk(
    35	      riskScore,
    36	      sellDays,
    37	      capital
    38	    ),
    39	    summary: buildSummary({
    40	      riskScore,
    41	      sellDays,
    42	      capital,
    43	    }),
    44	  };
    45	}
    46	
    47	function getInventoryRiskLabel(score) {
    48	  if (score <= 25) return "LOW_RISK";
    49	  if (score <= 50) return "MODERATE_RISK";
    50	  if (score <= 75) return "HIGH_RISK";
    51	  return "CRITICAL_RISK";
    52	}
    53	
    54	function getImmobilizedCapitalRisk(score, sellDays, capital) {
    55	  if (score >= 75) return "Crítico";
    56	  if (score >= 50) return "Alto";
    57	  if (sellDays <= 30 && capital <= 30000) return "Bajo";
    58	  return "Medio";
    59	}
    60	
    61	function buildSummary({
    62	  riskScore,
    63	  sellDays,
    64	  capital,
    65	}) {
    66	  return `Riesgo de inventario ${riskScore}/100. Capital estimado inmovilizado ${formatMoney(
    67	    capital
    68	  )} durante aproximadamente ${sellDays} días.`;
    69	}
    70	
    71	function formatMoney(value) {
    72	  return `${safeNumber(value).toLocaleString("es-ES")} €`;
    73	}
    74	
    75	function safeNumber(value) {
    76	  const number = Number(value);
    77	  return Number.isFinite(number) ? number : 0;
    78	}
    79	
    80	function clamp(value, min, max) {
    81	  return Math.min(Math.max(value, min), max);
    82	}
