     1	export function buildMarketTiming({
     2	  successProbability = 0,
     3	  sellSpeedScore = 50,
     4	  capitalEfficiencyScore = 50,
     5	  inventoryRiskScore = 50,
     6	  confidenceScore = 50,
     7	} = {}) {
     8	  let score = 0;
     9	
    10	  score += successProbability * 0.35;
    11	  score += sellSpeedScore * 0.25;
    12	  score += capitalEfficiencyScore * 0.20;
    13	  score += confidenceScore * 0.10;
    14	  score += Math.max(0, 100 - inventoryRiskScore) * 0.10;
    15	
    16	  score = clamp(Math.round(score), 0, 100);
    17	
    18	  return {
    19	    marketTimingScore: score,
    20	    marketTimingLabel: getLabel(score),
    21	    summary: buildSummary(score),
    22	  };
    23	}
    24	
    25	function getLabel(score) {
    26	  if (score >= 85) return "STRONG_ENTRY";
    27	  if (score >= 70) return "GOOD_ENTRY";
    28	  if (score >= 55) return "WAIT";
    29	  return "AVOID";
    30	}
    31	
    32	function buildSummary(score) {
    33	  if (score >= 85) {
    34	    return "Momento de entrada muy favorable.";
    35	  }
    36	
    37	  if (score >= 70) {
    38	    return "Buen momento para estudiar compra.";
    39	  }
    40	
    41	  if (score >= 55) {
    42	    return "Conviene esperar confirmación adicional.";
    43	  }
    44	
    45	  return "Momento de entrada poco favorable.";
    46	}
    47	
    48	function clamp(value, min, max) {
    49	  return Math.min(Math.max(value, min), max);
    50	}
