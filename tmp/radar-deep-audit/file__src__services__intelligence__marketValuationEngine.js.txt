     1	export function buildMarketValuation(vehicle = {}, options = {}) {
     2	  const price = toNumber(vehicle.price);
     3	
     4	  const memoryValuation =
     5	    vehicle.vehicleValuation ||
     6	    vehicle.memoryValuation ||
     7	    options.vehicleValuation ||
     8	    null;
     9	
    10	  const memoryEstimatedValue = toNumber(memoryValuation?.estimatedMarketValue);
    11	  const memoryConfidence = toNumber(memoryValuation?.confidence);
    12	  const memoryComparableCount = toNumber(memoryValuation?.comparableCount);
    13	
    14	  const estimatedMarketValue =
    15	    memoryEstimatedValue > 0 && memoryConfidence >= 50
    16	      ? memoryEstimatedValue
    17	      : buildFallbackMarketValue(vehicle);
    18	
    19	  // Calcular profit y roi basados en el valor de mercado estimado
    20	  const profit = Math.round(estimatedMarketValue - price);
    21	  const roi =
    22	    price > 0 ? Number(((profit / price) * 100).toFixed(2)) : 0;
    23	
    24	  const discountValue = profit;
    25	  const discountPercent = roi;
    26	
    27	  const valuationScore = calculateValuationScore({
    28	    discountPercent,
    29	    roi,
    30	    profit,
    31	    memoryConfidence,
    32	    memoryComparableCount,
    33	  });
    34	
    35	  return {
    36	    price,
    37	    estimatedMarketValue,
    38	    profit,
    39	    roi,
    40	    discountValue,
    41	    discountPercent,
    42	    valuationScore,
    43	    valuationLabel: buildValuationLabel(valuationScore),
    44	    valuationSource:
    45	      memoryEstimatedValue > 0 && memoryConfidence >= 50
    46	        ? "memory_comparables"
    47	        : "fallback_market_estimate",
    48	    valuationConfidence: memoryConfidence,
    49	    comparableCount: memoryComparableCount,
    50	  };
    51	}

    53	function buildFallbackMarketValue(vehicle = {}) {
    54	  const price = toNumber(vehicle.price);
    55	
    56	  if (price <= 0) {
    57	    return 0;
    58	  }
    59	
    60	  // Sin comparables reales, asumimos un valor conservador cercano al precio
    61	  return Math.round(price * 1.05);
    62	}

    64	function calculateValuationScore({
    65	  discountPercent,
    66	  roi,
    67	  profit,
    68	  memoryConfidence,
    69	  memoryComparableCount,
    70	}) {
    71	  let score = 50;
    72	
    73	  score += discountPercent * 2;
    74	  score += roi * 2;
    75	
    76	  if (profit > 0) {
    77	    score += 10;
    78	  }
    79	
    80	  if (profit >= 1000) {
    81	    score += 5;
    82	  }
    83	
    84	  if (profit >= 2000) {
    85	    score += 5;
    86	  }
    87	
    88	  if (memoryComparableCount >= 1) {
    89	    score += 5;
    90	  }
    91	
    92	  if (memoryComparableCount >= 3) {
    93	    score += 5;
    94	  }
    95	
    96	  if (memoryConfidence >= 70) {
    97	    score += 6;
    98	  }
    99	
   100	  if (memoryConfidence >= 85) {
   101	    score += 6;
   102	  }
   103	
   104	  if (memoryConfidence > 0 && memoryConfidence < 50) {
   105	    score -= 10;
   106	  }
   107	
   108	  if (roi < 0) {
   109	    score -= 12;
   110	  }
   111	
   112	  if (profit < 0) {
   113	    score -= 12;
   114	  }
   115	
   116	  return Math.max(0, Math.min(100, Math.round(score)));
   117	}

   119	function buildValuationLabel(score) {
   120	  if (score >= 85) {
   121	    return "Muy infravalorado";
   122	  }
   123	
   124	  if (score >= 70) {
   125	    return "Infravalorado";

   127
