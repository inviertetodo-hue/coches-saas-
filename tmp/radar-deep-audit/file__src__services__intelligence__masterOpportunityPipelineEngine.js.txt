   224	function sortModernOpportunities(a, b) {
   225	  const comparisons = [
   226	    ["allocationScore"],
   227	    ["executiveBuySignalScore"],
   228	    ["marketTimingScore"],
   229	    ["capitalEfficiencyScore"],
   230	    ["timelineMomentumScore"],
   231	    ["sellSpeedScore"],
   232	    ["successProbability"],
   233	    ["decision", "decisionScore"],
   234	    ["opportunity", "scoreV3"],
   235	    ["opportunity", "scoreV2"],
   236	  ];
   237	
   238	  for (const path of comparisons) {
   239	    const aValue = getNestedNumber(a, path);
   240	    const bValue = getNestedNumber(b, path);
   241	
   242	    if (bValue !== aValue) {
   243	      return bValue - aValue;
   244	    }
   245	  }
   246	
   247	  return Number(b.profit || 0) - Number(a.profit || 0);
   248	}

   252	  return Number(value || 0);

   352	function buildOpportunityScoreV2({
   353	  vehicle,
   354	  baseOpportunity,
   355	  comparables,
   356	  vehicleValuation,
   357	  roi = 0,
   358	  profit = 0,
   359	}) {
   360	  const baseScore = Number(
   361	    baseOpportunity?.score ??
   362	      baseOpportunity?.opportunityScore ??
   363	      baseOpportunity?.totalScore ??
   364	      0
   365	  );
   366	
   367	  const comparableCount = Number(comparables?.totalComparables || 0);
   368	  const confidence = Number(vehicleValuation?.confidence || 0);
   369	  const discountPercent = Number(vehicleValuation?.discountPercent || 0);
   370	
   371	  let score = baseScore || 40;
   372	
   373	  score += Math.min(18, Math.max(-18, roi * 2));
   374	  score += Math.min(15, Math.max(-15, discountPercent * 1.4));
   375	
   376	  if (profit > 0) score += 8;
   377	  if (profit >= 1000) score += 6;
   378	  if (profit >= 2000) score += 6;
   379	
   380	  if (comparableCount >= 1) score += 5;
   381	  if (comparableCount >= 2) score += 5;
   382	  if (comparableCount >= 5) score += 5;
   383	
   384	  if (confidence >= 60) score += 5;
   385	  if (confidence >= 75) score += 7;
   386	  if (confidence >= 90) score += 5;
   387	
   388	  if (confidence < 50) score -= 15;
   389	  if (comparableCount === 0) score -= 10;
   390	  if (roi < 0) score -= 18;
   391	  if (profit < 0) score -= 18;
   392	
   393	  let scoreV2 = clampScore(score);
   394	
   395	  // Cortafuegos económico del Radar
   396	  if (profit < 0 && roi <= 0) {
   397	    scoreV2 = Math.min(scoreV2, 35);
   398	  } else if (profit < 0) {
   399	    scoreV2 = Math.min(scoreV2, 45);
   400	  } else if (roi <= 0) {
   401	    scoreV2 = Math.min(scoreV2, 55);
   402	  }
   403	
   404	  return {
   405	    ...baseOpportunity,
   406	    scoreV2,
   407	    opportunityScoreV2: scoreV2,
   408	    opportunityLevelV2: buildOpportunityLevel(scoreV2),
   409	    opportunityReasonsV2: buildOpportunityReasons({
   410	      roi,
   411	      profit,
   412	      comparableCount,
   413	      confidence,
   414	      discountPercent,
   415	      scoreV2,
   416	    }),
   417	    valuationConfidence: confidence,
   418	    comparableCount,
   419	    discountPercent,
   420	    estimatedMarketValue: vehicleValuation?.estimatedMarketValue || 0,
   421	  };
   422	}

   424	function buildTimelineMomentum(timeline = null) {
   425	  if (!timeline || !Array.isArray(timeline.events) || timeline.events.length < 2) {
   426	    return {
   427	      timelineMomentumScore: 50,
   428	      timelineMomentumLabel: "NO_HISTORY",
   429	      timelineMomentumSummary: "Sin histórico temporal suficiente.",
   430	    };
   431	  }
   432	
   433	  const first = timeline.events[0];
   434	  const latest = timeline.events[timeline.events.length - 1];
   435	
   436	  let score = 50;
   437	
   438	  if (first.action !== "BUY" && latest.action === "BUY") score += 20;
   439	  if (first.action === "BUY" && latest.action !== "BUY") score -= 20;
   440	
   441	  const roiChange = Number(latest.roi || 0) - Number(first.roi || 0);
   442	  const profitChange = Number(latest.profit || 0) - Number(first.profit || 0);
   443	  const priceChange = Number(latest.price || 0) - Number(first.price || 0);
   444	  const sellSpeedChange =
   445	    Number(latest.sellSpeedScore || 0) - Number(first.sellSpeedScore || 0);
   446	
   447	  if (roiChange >= 5) score += 12;
   448	  if (roiChange <= -5) score -= 12;
   449	
   450	  if (profitChange >= 1000) score += 12;
   451	  if (profitChange <= -1000) score -= 12;
   452	
   453	  if (priceChange < 0) score += 8;
   454	  if (priceChange > 0) score -= 5;
   455	
   456	  if (sellSpeedChange >= 10) score += 8;
   457	  if (sellSpeedChange <= -10) score -= 8;
   458	
   459	  const timelineMomentumScore = clampScore(score);
   460	
   461	  return {
   462	    timelineMomentumScore,
   463	    timelineMomentumLabel: buildTimelineMomentumLabel(timelineMomentumScore),
   464	    timelineMomentumSummary: timeline.timelineSummary,
   465	  };
   466	}

   498	  }
   499

   502	  }
   503	
   504	  if (roi > 0) {
   505	    reasons.push(`ROI positivo estimado: ${roi}%.`);
   506	  }
   507	
   508	  if (profit > 0) {
   509	    reasons.push(`Margen positivo estimado: ${formatNumber(profit)} €.`);
   510	  }
   511	
   512	  if (discountPercent > 0) {
   513	    reasons.push(`Descuento frente a valoración: ${discountPercent}%.`);
   514	  }
   515	
   516	  if (comparableCount > 0) {
   517	    reasons.push(`${comparableCount} comparable(s) encontrados en memoria.`);
   518	  }
   519	
   520	  if (confidence >= 75) {
   521	    reasons.push(`Confianza de valoración alta: ${confidence}/100.`);
   522	  }
   523	
   524	  if (confidence < 50) {

   526	  }
   527	
   528	  if (comparableCount === 0) {
   529	    reasons.push("Sin comparables suficientes en memoria.");
   530	  }
   531	
   532	  return reasons;
   533	}

   535	function buildValuationSummary(records = []) {
   536	  const totalComparables = records.reduce(
   537	    (sum, item) => sum + Number(item.comparables?.totalComparables || 0),
   538	    0
   539	  );
   540	
   541	  const confidenceValues = records
   542	    .map((item) => Number(item.vehicleValuation?.confidence || 0))
   543	    .filter((value) => value > 0);
   544	
   545	  const discountValues = records
   546	    .map((item) => Number(item.vehicleValuation?.discountPercent || 0))
   547	    .filter((value) => Number.isFinite(value));
   548	
   549	  const scoreV2Values = records
   550	    .map((item) => Number(item.opportunity?.scoreV2 || 0))
   551	    .filter((value) => value > 0);
   552	
   553	  const sellSpeedValues = records
   554	    .map((item) => Number(item.sellSpeed?.sellSpeedScore || 0))
   555	    .filter((value) => value > 0);
   556	
   557	  const sellDaysValues = records
   558	    .map((item) => Number(item.sellSpeed?.estimatedSellDays || 0))
   559	    .filter((value) => value > 0);
   560	
   561	  const executiveValues = records
   562	    .map((item) => Number(item.executiveBuySignalScore || 0))
   563	    .filter((value) => value > 0);
   564	
   565	  const timingValues = records
   566	    .map((item) => Number(item.marketTimingScore || 0))
   567	    .filter((value) => value > 0);
   568	
   569	  const capitalEfficiencyValues = records
   570	    .map((item) => Number(item.capitalEfficiencyScore || 0))
   571	    .filter((value) => value > 0);
   572	
   573	  const timelineMomentumValues = records
   574	    .map((item) => Number(item.timelineMomentumScore || 0))
   575	    .filter((value) => value > 0);
   576	
   577	  return {
   578	    totalComparables,
   579	    averageConfidence: average(confidenceValues),
   580	    averageDiscountPercent: average(discountValues),
   581	    averageOpportunityScoreV2: average(scoreV2Values),
   582	    averageSellSpeedScore: average(sellSpeedValues),
   583	    averageEstimatedSellDays: average(sellDaysValues),
   584	    averageExecutiveBuySignalScore: average(executiveValues),
   585	    averageMarketTimingScore: average(timingValues),
   586	    averageCapitalEfficiencyScore: average(capitalEfficiencyValues),
   587	    averageTimelineMomentumScore: average(timelineMomentumValues),
   588	    vehiclesWithComparables: records.filter(
   589	      (item) => Number(item.comparables?.totalComparables || 0) > 0
   590	    ).length,
   591	  };
   592	}

   618	function clampScore(value) {
