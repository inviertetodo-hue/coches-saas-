   227	function calculateDecisionScoreV2({
   228	  opportunityScoreV2,
   229	  valuationScore,
   230	  qualityScore,
   231	  comparableConfidence,
   232	  comparableCount,
   233	  sellSpeedScore,
   234	  roi,
   235	  profit,
   236	  discountPercent,
   237	  executiveBuySignalScore,
   238	  timelineMomentumScore,
   239	  marketTimingScore,
   240	  capitalEfficiencyScore,
   241	  inventoryRiskScore,
   242	  allocationScore,
   243	  successProbability,
   244	}) {
   245	  let score = opportunityScoreV2 || 0;
   246	
   247	  if (!score) {
   248	    score = Math.round(
   249	      valuationScore * 0.30 +
   250	        qualityScore * 0.16 +
   251	        comparableConfidence * 0.22 +
   252	        sellSpeedScore * 0.16 +
   253	        successProbability * 0.16
   254	    );
   255	  }
   256	
   257	  if (executiveBuySignalScore > 0) {
   258	    score = Math.round(score * 0.72 + executiveBuySignalScore * 0.28);
   259	  }
   260	
   261	  if (allocationScore > 0) {
   262	    score = Math.round(score * 0.86 + allocationScore * 0.14);
   263	  }
   264	
   265	  if (marketTimingScore > 0) {
   266	    score = Math.round(score * 0.90 + marketTimingScore * 0.10);
   267	  }
   268	
   269	  if (capitalEfficiencyScore > 0) {
   270	    score = Math.round(score * 0.91 + capitalEfficiencyScore * 0.09);
   271	  }
   272	
   273	  if (timelineMomentumScore > 0) {
   274	    score += Math.round((timelineMomentumScore - 50) * 0.18);
   275	  }
   276	
   277	  if (inventoryRiskScore >= 80) score -= 18;
   278	  if (inventoryRiskScore >= 65) score -= 10;
   279	
   280	  if (sellSpeedScore < 50 && sellSpeedScore > 0) score -= 8;
   281	
   282	  if (discountPercent < 0) score -= 12;
   283	
   284	  if (roi < 0) score -= 15;
   285	  if (profit < 0) score -= 15;
   286	
   287	  if (qualityScore > 0 && qualityScore < 60) score -= 20;
   288	  if (comparableConfidence > 0 && comparableConfidence < 55) score -= 12;
   289	
   290	  let decisionScore = clampScore(score);
   291	
   292	  // Cortafuegos económico del Radar
   293	  if (profit < 0 && roi <= 0) {
   294	    decisionScore = Math.min(decisionScore, 35);
   295	  } else if (profit < 0) {
   296	    decisionScore = Math.min(decisionScore, 45);
   297	  } else if (roi <= 0) {
   298	    decisionScore = Math.min(decisionScore, 55);
   299	  }
   300	
   301	  // Techo económico: reservar 95-100 para oportunidades excepcionales.
   302	  if (roi > 0 && roi < 3) {
   303	    decisionScore = Math.min(decisionScore, 60);
   304	  } else if (roi >= 3 && roi < 5) {
   305	    decisionScore = Math.min(decisionScore, 75);
   306	  } else if (roi >= 5 && roi < 8) {
   307	    decisionScore = Math.min(decisionScore, 85);
   308	  } else if (roi >= 8 && roi < 12) {
   309	    decisionScore = Math.min(decisionScore, 92);
   310	  }
   311	
   312	  if (profit > 0 && profit < 1000) {
   313	    decisionScore = Math.min(decisionScore, 62);
   314	  } else if (profit >= 1000 && profit < 2000) {
   315	    decisionScore = Math.min(decisionScore, 76);
   316	  } else if (profit >= 2000 && profit < 4000) {
   317	    decisionScore = Math.min(decisionScore, 88);
   318	  }
   319	
   320	  return decisionScore;
   321	}

   323	function buildActionV2({
   324	  decisionScore,
   325	  opportunityScoreV2,
   326	  qualityScore,
   327	  comparableConfidence,
   328	  comparableCount,
   329	  roi,
   330	  profit,
   331	  executiveBuySignalScore,
   332	  executiveBuySignalLabel,
   333	  timelineMomentumScore,
   334	  timelineMomentumLabel,
   335	  marketTimingScore,
   336	  marketTimingLabel,
   337	  capitalEfficiencyScore,
   338	  inventoryRiskScore,
   339	  inventoryRiskLabel,
   340	  allocationScore,
   341	  allocationTier,
   342	  successProbability,
   343	}) {
   344	  if (qualityScore > 0 && qualityScore < 60) {
   345	    return { action: "REJECT", label: "Descartar" };
   346	  }
   347	
   348	  if (roi < 0 || profit < 0) {
   349	    return { action: "WATCH", label: "Observar mercado" };
   350	  }
   351	
   352	  if (
   353	    inventoryRiskLabel === "CRITICAL_RISK" ||
   354	    inventoryRiskScore >= 85 ||
   355	    marketTimingLabel === "AVOID" ||
   356	    timelineMomentumLabel === "AVOID_TREND"
   357	  ) {
   358	    return { action: "REJECT", label: "Descartar" };
   359	  }
   360	
   361	  const hasExecutiveBuy =
   362	    executiveBuySignalLabel === "STRONG_BUY" ||
   363	    executiveBuySignalLabel === "BUY" ||
   364	    executiveBuySignalScore >= 78;
   365	
   366	  const hasStrongScore = opportunityScoreV2 >= 85 && decisionScore >= 80;
   367	  const hasReliableEvidence =
   368	    comparableConfidence >= 70 && comparableCount >= 2;
   369	  const hasPositiveEconomics =
   370	    roi >= 5 &&
   371	    profit >= 1500;
   372	
   373	  const hasModernConfirmation =
   374	    successProbability >= 70 ||
   375	    allocationScore >= 70 ||
   376	    allocationTier === "TIER_1" ||
   377	    allocationTier === "TIER_2" ||
   378	    capitalEfficiencyScore >= 70 ||
   379	    marketTimingScore >= 70 ||
   380	    timelineMomentumScore >= 70;
   381	
   382	  const hasModernRisk =
   383	    inventoryRiskScore >= 70 ||
   384	    marketTimingLabel === "WAIT" ||
   385	    timelineMomentumLabel === "DETERIORATING";
   386	
   387	  if (
   388	    hasExecutiveBuy &&
   389	    hasPositiveEconomics &&
   390	    decisionScore >= 75 &&
   391	    !hasModernRisk
   392	  ) {
   393	    return { action: "BUY", label: "Comprar" };
   394	  }
   395	
   396	  if (
   397	    hasStrongScore &&
   398	    hasReliableEvidence &&
   399	    hasPositiveEconomics &&
   400	    hasModernConfirmation &&
   401	    !hasModernRisk
   402	  ) {
   403	    return { action: "BUY", label: "Comprar" };
   404	  }
   405	
   406	  if (hasExecutiveBuy || hasStrongScore || hasModernConfirmation) {
   407	    return { action: "WATCH", label: "Validar oportunidad" };
   408	  }
   409	
   410	  if (opportunityScoreV2 >= 65 || decisionScore >= 65) {
   411	    return { action: "WATCH", label: "Vigilar / revisar" };
   412	  }
   413	
   414	  if (comparableCount === 0 && comparableConfidence < 60) {
   415	    return { action: "REJECT", label: "Descartar" };
   416	  }
   417	
   418	  return { action: "REJECT", label: "Descartar" };
   419	}

   421	function buildDecisionConfidenceV2({
   422	  qualityScore,
   423	  comparableConfidence,
   424	  opportunityScoreV2,
   425	  valuationScore,
   426	  comparableCount,
   427	  executiveBuySignalScore,
   428	  timelineMomentumScore,
   429	  marketTimingScore,
   430	  capitalEfficiencyScore,
   431	}) {
   432	  let score = Math.round(
   433	    qualityScore * 0.20 +
   434	      comparableConfidence * 0.25 +
   435	      opportunityScoreV2 * 0.20 +
   436	      valuationScore * 0.15 +
   437	      executiveBuySignalScore * 0.08 +
   438	      timelineMomentumScore * 0.04 +
   439	      marketTimingScore * 0.04 +
   440	      capitalEfficiencyScore * 0.04
   441	  );
   442	
   443	  if (comparableCount >= 1) score += 5;
   444	  if (comparableCount >= 3) score += 5;
   445	
   446	  score = clampScore(score);
   447	
   448	  if (score >= 85) {
   449	    return { level: "high", label: "Alta", score };
   450	  }
   451	
   452	  if (score >= 70) {
   453	    return { level: "medium", label: "Media", score };
   454	  }
   455	
   456	  return { level: "low", label: "Baja", score };
   457	}

   547	  }

   551	      `Eficiencia de capital: ${capitalEfficiencyLabel || "SIN_LABEL"} (${capitalEfficiencyScore}/100).`

   602	function buildRisksV2({
   603	  qualityScore,
   604	  comparableConfidence,
   605	  comparableCount,
   606	  roi,
   607	  profit,
   608	  discountPercent,
   609	  sellSpeedScore,
   610	  executiveBuySignalScore,
   611	  timelineMomentumScore,
   612	  marketTimingScore,
   613	  inventoryRiskScore,
   614	  allocationScore,
   615	}) {
   616	  const risks = [];
   617	
   618	  if (qualityScore > 0 && qualityScore < 70) {
   619	    risks.push("Calidad de datos por debajo del umbral recomendado.");
   620	  }
   621	
   622	  if (comparableCount === 0) {
   623	    risks.push("Sin comparables suficientes para defender la valoración.");
   624	  }
   625	
   626	  if (comparableCount > 0 && comparableCount < 2) {
   627	    risks.push("Pocos comparables para confirmar una compra directa.");
   628	  }
   629	
   630	  if (comparableConfidence > 0 && comparableConfidence < 70) {
   631	    risks.push("Confianza de valoración baja.");
   632	  }
   633	
   634	  if (roi < 0) {
   635	    risks.push("ROI negativo.");
   636	  }
   637	
   638	  if (profit < 0) {
   639	    risks.push("Margen estimado negativo.");
   640	  }
   641	
   642	  if (discountPercent <= 0) {
   643	    risks.push("No existe descuento claro frente al valor estimado.");
   644	  }
   645	
   646	  if (sellSpeedScore > 0 && sellSpeedScore < 55) {
   647	    risks.push("Riesgo de rotación lenta.");
   648	  }
   649	
   650	  if (executiveBuySignalScore > 0 && executiveBuySignalScore < 62) {
   651	    risks.push("Señal ejecutiva moderna todavía débil.");
   652	  }
   653	
   654	  if (timelineMomentumScore > 0 && timelineMomentumScore < 45) {
   655	    risks.push("Momentum temporal negativo.");
   656	  }
   657	
   658	  if (marketTimingScore > 0 && marketTimingScore < 50) {
   659	    risks.push("Timing de mercado poco favorable.");
   660	  }
   661	
   662	  if (inventoryRiskScore >= 70) {
   663	    risks.push("Riesgo de inventario elevado.");
   664	  }
   665	
   666	  if (allocationScore > 0 && allocationScore < 50) {
   667	    risks.push("Baja prioridad de asignación de capital.");
   668	  }
   669	
   670	  if (risks.length === 0) {
   671	    risks.push("No se detectan riesgos críticos en la evaluación inicial.");
   672	  }
   673	
   674	  return risks;
   675	}

   737	function normalizeNumber(value) {
   738	  const numericValue = Number(value);
   739	
   740	  if (!Number.isFinite(numericValue)) {
   741	    return 0;
   742	  }
   743	
   744	  return numericValue;
   745	}

   747	function clampScore(value) {
   748	  return Math.max(0, Math.min(100, Math.round(value)));
   749	}

   751	function formatCurrency(value) {
   752	  return `${Math.round(normalizeNumber(value)).toLocaleString("es-ES")} €`;
   753	}
