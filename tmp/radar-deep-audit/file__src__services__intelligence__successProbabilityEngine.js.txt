     1	export function buildSuccessProbability({
     2	  opportunity = {},
     3	  prediction = {},
     4	  learning = {},
     5	} = {}) {
     6	  const executiveScore = safeNumber(
     7	    opportunity.executiveScore ||
     8	      opportunity.priorityScore ||
     9	      opportunity.score
    10	  );
    11	
    12	  const roi = safeNumber(
    13	    opportunity.roi ??
    14	      prediction.expectedROI ??
    15	      opportunity.historicalAverageROI
    16	  );
    17	
    18	  const profit = safeNumber(
    19	    opportunity.profit ??
    20	      prediction.expectedProfit ??
    21	      opportunity.historicalAverageProfit
    22	  );
    23	
    24	  const confidenceScore = safeNumber(
    25	    opportunity.historicalConfidenceScore ??
    26	      prediction.confidenceScore ??
    27	      learning.confidenceScore
    28	  );
    29	
    30	  const learningBonus = safeNumber(
    31	    opportunity.learningBonus ||
    32	      learning.learningBonus
    33	  );
    34	
    35	  let successProbability = 0;
    36	
    37	  successProbability += executiveScore * 0.22;
    38	  successProbability += buildRoiScore(roi) * 0.36;
    39	  successProbability += buildProfitScore(profit) * 0.32;
    40	  successProbability += confidenceScore * 0.08;
    41	  successProbability += learningBonus * 0.8;
    42	
    43	  if (roi >= 8 && profit >= 4000) successProbability += 12;
    44	  if (roi >= 10 && profit >= 5000) successProbability += 10;
    45	  if (roi <= 0 || profit <= 0) successProbability -= 18;
    46	  if (roi > 0 && roi < 3) successProbability -= 8;
    47	  if (profit > 0 && profit < 1000) successProbability -= 8;
    48	
    49	  successProbability = clamp(Math.round(successProbability), 0, 100);
    50	
    51	  return {
    52	    successProbability,
    53	    buySignal: getBuySignal(successProbability),
    54	    expectedROI: roi,
    55	    expectedProfit: profit,
    56	    confidenceScore,
    57	    learningBonus,
    58	    riskLabel: getRiskLabel(successProbability, confidenceScore),
    59	    summary: buildSummary({
    60	      successProbability,
    61	      roi,
    62	      profit,
    63	      confidenceScore,
    64	      learningBonus,
    65	    }),
    66	  };
    67	}

    90	function buildRoiScore(roi) {
    91	  if (roi <= 0) return 0;
    92	  if (roi >= 15) return 100;
    93	  if (roi >= 12) return 92;
    94	  if (roi >= 10) return 84;
    95	  if (roi >= 8) return 74;
    96	  if (roi >= 6) return 64;
    97	  if (roi >= 5) return 56;
    98	  if (roi >= 3) return 42;
    99	  return 24;
   100	}

   102	function buildProfitScore(profit) {
   103	  if (profit <= 0) return 0;
   104	  if (profit >= 8000) return 100;
   105	  if (profit >= 6000) return 88;
   106	  if (profit >= 5000) return 78;
   107	  if (profit >= 4000) return 68;
   108	  if (profit >= 3000) return 56;
   109	  if (profit >= 2000) return 44;
   110	  if (profit >= 1000) return 30;
   111	  return 16;
   112	}

   114	function getBuySignal(probability) {

   116	  if (probability >= 72) return "INTERESANTE";

   121	function getRiskLabel(probability, confidenceScore) {
   122	  if (confidenceScore < 50) return "Desconocido";
   123	  if (probability >= 85) return "Bajo";
   124	  if (probability >= 72) return "Medio-bajo";
   125	  if (probability >= 58) return "Medio";
   126	  return "Alto";

   139	function formatBonus(value) {

   145	function safeNumber(value) {
