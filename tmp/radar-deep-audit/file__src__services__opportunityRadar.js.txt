     1	export function generateOpportunityRadar(analyses = []) {
     2	  if (!Array.isArray(analyses) || analyses.length === 0) {
     3	    return {
     4	      radarScore: 0,
     5	      radarLevel: "Sin datos",
     6	      priorityOpportunities: [],
     7	      radarInsights: [],
     8	    };
     9	  }
    10	
    11	  const priorityOpportunities = analyses
    12	    .filter(Boolean)
    13	    .map(buildRadarOpportunity)
    14	    .filter((item) => item.action !== "REJECT")
    15	    .filter((item) => item.radarPriority >= 45)
    16	    .sort((a, b) => b.radarPriority - a.radarPriority)
    17	    .slice(0, 12);
    18	
    19	  const radarScore = calculateRadarScore(priorityOpportunities);
    20	  const radarLevel = buildRadarLevel(radarScore);
    21	  const radarInsights = buildRadarInsights(priorityOpportunities, radarScore);
    22	
    23	  return {
    24	    radarScore,
    25	    radarLevel,
    26	    priorityOpportunities,
    27	    radarInsights,
    28	  };
    29	}

   144	function calculateRadarPriority({
   145	  action,
   146	  decisionScore,
   147	  score,
   148	  roi,
   149	  profit,
   150	  confidence,
   151	  comparableCount,
   152	  sellSpeedScore,
   153	  estimatedSellDays,
   154	  discountPercent,
   155	  discoverySignals,
   156	}) {
   157	  let priority =
   158	    decisionScore * 0.42 +
   159	    score * 0.22 +
   160	    confidence * 0.14 +
   161	    sellSpeedScore * 0.12;
   162	
   163	  if (roi > 0) priority += Math.min(9, roi * 0.28);
   164	  if (profit > 0) priority += Math.min(9, profit / 1000);
   165	
   166	  if (discountPercent > 0) priority += Math.min(10, discountPercent * 0.9);
   167	
   168	  if (comparableCount >= 2) priority += 5;
   169	  if (comparableCount >= 5) priority += 4;
   170	
   171	  if (estimatedSellDays > 0 && estimatedSellDays <= 30) priority += 6;
   172	  if (estimatedSellDays > 60) priority -= 6;
   173	
   174	  if (action === "BUY") priority += 10;
   175	  if (action === "WATCH") priority += 3;
   176	
   177	  const strongSignals = discoverySignals.filter(
   178	    (signal) => signal.importance === "HIGH"
   179	  ).length;
   180	
   181	  priority += strongSignals * 4;
   182	
   183	  return clampScore(priority);
   184	}

   267	function calculateRadarScore(opportunities = []) {
   268	  if (!opportunities.length) return 0;
   269	
   270	  const average =
   271	    opportunities.reduce((acc, item) => acc + item.radarPriority, 0) /
   272	    opportunities.length;
   273	
   274	  return clampScore(average);
   275	}

   277	function buildRadarLevel(score) {
   278	  if (score >= 85) return "Alto";
   279	  if (score >= 65) return "Medio";

   283	function buildDiscoveryLevel(priority) {
   284	  if (priority >= 85) return "DISCOVER_NOW";
   285	  if (priority >= 70) return "HIGH_PRIORITY";
   286	  if (priority >= 55) return "WATCH_CLOSELY";
   287	  return "OBSERVE";

   290	function buildRadarInsights(opportunities = [], radarScore = 0) {
   291	  const insights = [];
   292	
   293	  const buyCount = opportunities.filter((item) => item.action === "BUY").length;
   294	  const fastSellCount = opportunities.filter(
   295	    (item) => item.sellSpeedScore >= 80 || item.estimatedSellDays <= 30
   296	  ).length;
   297	  const strongDiscoveryCount = opportunities.filter(
   298	    (item) => item.discoveryLevel === "DISCOVER_NOW"
   299	  ).length;
   300	
   301	  if (buyCount > 0) {
   302	    insights.push(`🚀 ${buyCount} oportunidad(es) BUY detectadas por el motor.`);
   303	  }
   304	
   305	  if (strongDiscoveryCount > 0) {
   306	    insights.push(`🔥 ${strongDiscoveryCount} oportunidad(es) requieren revisión inmediata.`);
   307	  }
   308	
   309	  if (fastSellCount > 0) {
   310	    insights.push(`⚡ ${fastSellCount} candidato(s) tienen señal de rotación rápida.`);
   311	  }
   312	
   313	  if (radarScore >= 85) {
   314	    insights.push("🏆 El radar detecta un mercado especialmente atractivo.");
   315	  }
   316	
   317	  if (!opportunities.length) {
   318	    insights.push("📡 No existen oportunidades válidas tras aplicar los filtros del motor.");
   319	  }
   320	
   321	  return insights;
   322	}

   344	function clampScore(value) {
   345	  if (!Number.isFinite(Number(value))) return 0;
   346	  return Math.max(0, Math.min(100, Math.round(value)));
