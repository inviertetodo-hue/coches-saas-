   141	function buildTimelineMomentum(timeline = null) {
   142	  if (!timeline || !Array.isArray(timeline.events) || timeline.events.length < 2) {
   143	    return {
   144	      timelineMomentumScore: 50,
   145	      timelineMomentumLabel: "NO_HISTORY",
   146	      timelineMomentumSummary: "Sin histórico temporal suficiente.",
   147	    };
   148	  }
   149	
   150	  const first = timeline.events[0];
   151	  const latest = timeline.events[timeline.events.length - 1];
   152	
   153	  let score = 50;
   154	
   155	  if (first.action !== "BUY" && latest.action === "BUY") score += 20;
   156	  if (first.action === "BUY" && latest.action !== "BUY") score -= 20;
   157	
   158	  const roiChange = safeNumber(latest.roi) - safeNumber(first.roi);
   159	  const profitChange = safeNumber(latest.profit) - safeNumber(first.profit);
   160	  const priceChange = safeNumber(latest.price) - safeNumber(first.price);
   161	  const sellSpeedChange =
   162	    safeNumber(latest.sellSpeedScore) - safeNumber(first.sellSpeedScore);
   163	
   164	  if (roiChange >= 5) score += 12;
   165	  if (roiChange <= -5) score -= 12;
   166	
   167	  if (profitChange >= 1000) score += 12;
   168	  if (profitChange <= -1000) score -= 12;
   169	
   170	  if (priceChange < 0) score += 8;
   171	  if (priceChange > 0) score -= 5;
   172	
   173	  if (sellSpeedChange >= 10) score += 8;
   174	  if (sellSpeedChange <= -10) score -= 8;
   175	
   176	  const timelineMomentumScore = clampScore(score);
   177	
   178	  return {
   179	    timelineMomentumScore,
   180	    timelineMomentumLabel: buildTimelineMomentumLabel(timelineMomentumScore),
   181	    timelineMomentumSummary: timeline.timelineSummary,
   182	  };
   183	}

   185	function buildTimelineMomentumLabel(score) {
   186	  if (score >= 80) return "IMPROVING_FAST";
   187	  if (score >= 65) return "IMPROVING";

   237	function calculateRankingScore(topOpportunities = []) {
   238	  if (topOpportunities.length === 0) return 0;
   239	
   240	  return Math.round(
   241	    topOpportunities.reduce((sum, item) => {
   242	      return (
   243	        sum +
   244	        Math.round(
   245	          safeNumber(item.allocationScore) * 0.35 +
   246	            safeNumber(item.executiveBuySignalScore) * 0.22 +
   247	            safeNumber(item.marketTimingScore) * 0.15 +
   248	            safeNumber(item.capitalEfficiencyScore) * 0.13 +
   249	            safeNumber(item.timelineMomentumScore) * 0.15
   250	        )
   251	      );
   252	    }, 0) / topOpportunities.length
   253	  );
   254	}

   487	function clampScore(value) {
